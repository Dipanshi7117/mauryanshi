Ques-22 Write a program to calculate area and perimeter of rectangle ?

Code

#include<stdio.h>

int main()
{
    int l, b, area, perimeter;

    printf("Enter length: ");
    scanf("%d", &l);

    printf("Enter breadth: ");
    scanf("%d", &b);

    area = l * b;
    perimeter = 2 * (l + b);

    printf("Area of Rectangle = %d\n", area);
    printf("Perimeter of Rectangle = %d", perimeter);

    return 0;
}

Output

Enter length: 5
Enter breadth: 3

Area of Rectangle = 15
Perimeter of Rectangle = 16
