Ques 4 Enter a file name from user and copy that file in another file.

Code

#include<stdio.h>
#include<conio.h>

void main()
{
    FILE *a, *b;
    char name1[20], name2[20], ch;

    clrscr();

    printf("Enter first file name: ");
    scanf("%s",name1);

    printf("Enter second file name: ");
    scanf("%s",name2);

    a=fopen(name1,"r");
    b=fopen(name2,"w");

    while((ch=getc(a))!=EOF)
    {
        putc(ch,b);
    }

    fclose(a);
    fclose(b);

    printf("File copied successfully.");

getch();
}

Output

Enter first file name: abc.txt
Enter second file name: xyz.txt

File copied successfully.

Ques 6 Enter two files from user and concatenate into 3rd file.

Code

#include<stdio.h>
#include<conio.h>

void main()
{
    FILE *a, *b, *c;
    char name1[20], name2[20], name3[20], ch;

    clrscr();

    printf("Enter first file name: ");
    scanf("%s",name1);

    printf("Enter second file name: ");
    scanf("%s",name2);

    printf("Enter third file name: ");
    scanf("%s",name3);

    a=fopen(name1,"r");
    b=fopen(name2,"r");
    c=fopen(name3,"w");

    while((ch=getc(a))!=EOF)
        putc(ch,c);

    while((ch=getc(b))!=EOF)
        putc(ch,c);

    fclose(a);
    fclose(b);
    fclose(c);

    printf("Files concatenated successfully.");

    getch();
}

Output

Enter first file name: abc.txt
Enter second file name: xyz.txt
Enter third file name: pqr.txt

Files concatenated successfully.
