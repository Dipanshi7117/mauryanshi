Que-3 Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form ? Start Solving

Code

#include <stdio.h>

struct Employee
{
    int ID;
    char name[50];
    char designation[50];
};

int main()
{
    struct Employee emp[50];
    int i, j;

    for(i = 0; i < 50; i++)
    {
        printf("\nEnter details of Employee %d\n", i+1);

        printf("Enter ID: ");
        scanf("%d", &emp[i].ID);

        printf("Enter Name: ");
        scanf("%49s", emp[i].name);

        printf("Enter Designation: ");
        scanf("%49s", emp[i].designation);
    }
    printf("\n\nEMPLOYEE DETAILS\n");
    printf("ID\tNAME CHARACTERS\n");

    for(i = 0; i < 50; i++)
    {
        printf("%d\t", emp[i].ID);

        for(j = 0; emp[i].name[j] != '\0'; j++)
        {
            printf("%c ", emp[i].name[j]);
        }

        printf("\n");
    }

    return 0;
}

Que-4 Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?

Code

#include <stdio.h>

struct Student
{
    int roll;
    char name[50];
    float fee;
    char dob[20];
};

int main()
{
    struct Student s[50], temp;
    int i, j, n;

    printf("Enter number of students: ");
    scanf("%d", &n);

    // Input student details
    for(i = 0; i < n; i++)
    {
        printf("\nEnter details of student %d\n", i+1);

        printf("Enter Roll Number: ");
        scanf("%d", &s[i].roll);

        printf("Enter Name: ");
        scanf("%49s", s[i].name);

        printf("Enter Fee: ");
        scanf("%f", &s[i].fee);

        printf("Enter Date of Birth (DD/MM/YYYY): ");
        scanf("%19s", s[i].dob);
    }

    // Sort students in ascending order of roll number
    for(i = 0; i < n-1; i++)
    {
        for(j = 0; j < n-i-1; j++)
        {
            if(s[j].roll > s[j+1].roll)
            {
                temp = s[j];
                s[j] = s[j+1];
                s[j+1] = temp;
            }
        }
    }

    // Display student details
    printf("\n\nSTUDENT DETAILS IN ASCENDING ORDER\n");

    printf("----------------------------------------------------------\n");
    printf("ROLL\tNAME\tFEE\t\tDOB\n");
    printf("----------------------------------------------------------\n");

    for(i = 0; i < n; i++)
    {
        printf("%d\t%s\t%.2f\t%s\n",
               s[i].roll,
               s[i].name,
               s[i].fee,
               s[i].dob);
    }

    return 0;
}
