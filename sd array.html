Que-2  Enter an array from user and print the array in reverse order ?


Code:


#include <stdio.h>

int main() {
    int arr[100], n, i;

    printf("Enter the size of array: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for(i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    printf("Array in reverse order:\n");

    for(i = n - 1; i >= 0; i--) {
        printf("%d ", arr[i]);
    }

    return 0;
}




Output


Enter the size of array: 5

Enter array elements:
10
20
30
40
50

Array in reverse order:
50 40 30 20 10





Que-3  Enter an array from user and copy the array into second array ?


Code

int main() {
    int arr1[100], arr2[100], n, i;

    printf("Enter the size of array: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for(i = 0; i < n; i++) {
        scanf("%d", &arr1[i]);
    }

    // Copy first array into second array
    for(i = 0; i < n; i++) {
        arr2[i] = arr1[i];
    }

    printf("First array:\n");

    for(i = 0; i < n; i++) {
        printf("%d ", arr1[i]);
    }

    printf("\nSecond array (copied):\n");

    for(i = 0; i < n; i++) {
        printf("%d ", arr2[i]);
    }

    return 0;
}

{

    printf("Enter 5 array elements:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &arr[i]);
    }

    printf("Enter a number to search: ");
    scanf("%d", &num);

    for(i = 0; i < 5; i++)
    {
        if(arr[i] == num)
        {
            found = 1;
            break;
        }
    }

    if(found == 1)
    {
        printf("Found");
    }
    else
    {
        printf("Not Found");
    }

    return 0;
}

Que-12  Find maximum values of array ?


Code



#include <stdio.h>

int main()
{
    int arr[5], i, max;

    printf("Enter 5 array elements:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &arr[i]);
    }

    max = arr[0];

    for(i = 1; i < 5; i++)
    {
        if(arr[i] > max)
        {
            max = arr[i];
        }
    }

    printf("Maximum value = %d\n", max);

    return 0;
}





Output


Enter 5 array elements:
10
50
20
80
30

Maximum value = 80
