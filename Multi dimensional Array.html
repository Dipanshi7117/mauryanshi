Que-1 Enter the four number from user and print it into matrix

Code


#include <stdio.h>

int main()
{
    int arr[2][2], i, j;

    printf("Enter 4 numbers:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &arr[i][j]);
        }
    }

    printf("Matrix is:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", arr[i][j]);
        }

        printf("\n");
    }

    return 0;
}

    {

        }
    }

    printf("Enter second matrix elements:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    // Addition of two matrices
    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {

    printf("Addition of two matrices:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {

Que-5 Enter 2 matrix from user and multiply them ?

Code


#include <stdio.h>

int main()
{
    int a[2][2], b[2][2], c[2][2];
    int i, j, k;

    printf("Enter first matrix elements:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix elements:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    // Multiplication of two matrices
    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            c[i][j] = 0;

            for(k = 0; k < 2; k++)
            {
                c[i][j] = c[i][j] + a[i][k] * b[k][j];
            }
        }
    }

    printf("Multiplication of two matrices:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", c[i][j]);
        }
        printf("\n");
    }

    return 0;
}
