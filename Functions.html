Que-2 Write a program to enter 10 number from the user add the number and display the answer ?

Code


#include <stdio.h>

void sum()
{
    int i, num, total = 0;

    for(i = 1; i <= 10; i++)
    {
        printf("Enter number %d: ", i);
        scanf("%d", &num);

        total = total + num;
    }

    printf("Sum of 10 numbers = %d", total);
}

int main()
{
    sum();

    return 0;

Que-6  Swap numbers using Call by function ?


Code


#include 

void swap(int a, int b)
{
    int temp;

    temp = a;
    a = b;
    b = temp;

    printf("After swapping: a = %d, b = %d\n", a, b);
}

int main()
{
    int x, y;

    printf("Enter two numbers: ");
    scanf("%d %d", &x, &y);

    printf("Before swapping: x = %d, y = %d\n", x, y);

    swap(x, y);

    return 0;
}




Output


Enter two numbers: 10 20

Before swapping: x = 10, y = 20
After swapping: a = 20, b = 10




Que-7 Make function for find out power of number, enter the number and power from user ?


Code


#include <stdio.h>

int power(int num, int p)
{
    int i, result = 1;

    for(i = 1; i <= p; i++)
    {
        result = result * num;
    }

    return result;
}

int main()
{
    int num, p, result;

    printf("Enter a number: ");
    scanf("%d", &num);

    printf("Enter power: ");
    scanf("%d", &p);

    result = power(num, p);

    printf("Result = %d", result);

    return 0;
}




Output


Enter a number: 2
Enter power: 3
Result = 8



Que-8  Make function for find out Cube a user entered number ?


Code


#include <stdio.h>

int cube(int num)
{
    return num * num * num;
}

int main()
{
    int n, result;

    printf("Enter a number: ");
    scanf("%d", &n);

    result = cube(n);

    printf("Cube of %d = %d", n, result);

    return 0;
}
