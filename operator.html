Q-11   Enter marks of students and calculate average of student’s marks ?


Code


#include <stdio.h>
#include <conio.h>
	
void main()
{
   
    int m1, m2, m3, total;
    float average;

    clrscr();

    printf("Enter marks of first subject: ");
    scanf("%d", &m1);

    printf("Enter marks of second subject: ");
    scanf("%d", &m2);

    printf("Enter marks of third subject: ");
    scanf("%d", &m3);

    total = m1 + m2 + m3;

    average = total / 3.0;

    printf("\nTotal marks = %d", total);
    printf("\nAverage marks = %.2f", average);

    getch();
}





Output


Enter marks of first subject: 80
Enter marks of second subject: 70
Enter marks of third subject: 90

Total marks = 240
Average marks = 80.00




Q-12   Calculate simple interest = (P x R x T)/ 100


Code


#include <stdio.h>
#include <conio.h>
	
void main()
{
     float p, r, t, si;

    clrscr();

    printf("Enter Principal Amount: ");
    scanf("%f", &p);

    printf("Enter Rate of Interest: ");
    scanf("%f", &r);

    printf("Enter Time Period: ");
    scanf("%f", &t);

    si = (p * r * t) / 100;

    printf("\nSimple Interest = %.2f", si);

    getch();
}





Output


Enter Principal Amount: 10000
Enter Rate of Interest: 5
Enter Time Period: 2

Simple Interest = 1000.00
